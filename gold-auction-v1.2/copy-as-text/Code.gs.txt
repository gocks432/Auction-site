function hash_(password,salt){var bytes=Utilities.computeHmacSha256Signature(salt+'|'+password,props_().getProperty('SECRET_PEPPER'));return Utilities.base64Encode(bytes);}
function eq_(a,b){if(typeof a!=='string'||typeof b!=='string')return false;var diff=a.length^b.length;for(var i=0;i<Math.max(a.length,b.length);i++)diff|=(a.charCodeAt(i)||0)^(b.charCodeAt(i)||0);return diff===0;}

function login_(s,p){var teacher=p.role==='teacher',code=str_(p.code,32).toUpperCase(),key='attempt:'+ (teacher?'teacher':hash_(code,'login')),cache=CacheService.getScriptCache(),record=JSON.parse(cache.get(key)||'{"n":0,"until":0}');if(record.until>Date.now())fail_('로그인 시도가 많습니다. 10분 후 다시 시도해 주세요.');var u=teacher?s.teacher:s.students.find(function(x){return x.code===code;}),ok=u&&eq_(u.hash,hash_(String(p.password||''),u.salt));if(!ok){record.n++;if(record.n>=5)record.until=Date.now()+600000;cache.put(key,JSON.stringify(record),600);fail_('아이디 또는 비밀번호가 올바르지 않습니다.');}cache.remove(key);var token=uuid_()+uuid_(),ttl=teacher?7200:21600,a={role:teacher?'teacher':'student',id:teacher?'teacher':u.id,version:teacher?u.version:u.authVersion,expires:Date.now()+ttl*1000};cache.put('session:'+token,JSON.stringify(a),ttl);return {token:token,role:a.role};}

function updateStudent_(s,p){
  var u=student_(s,p.id),pin=String(p.pin||'').trim();
  var row=validateImport_(s,[{number:u.number,code:p.code,name:p.name,pin:pin||'0000'}])[0];
  var credentialsChanged=u.code!==row.code||!!pin;
  u.code=row.code;u.name=row.name;
  if(pin){u.salt=uuid_();u.hash=hash_(pin,u.salt);}
  if(credentialsChanged)u.authVersion++;
}
